  arr = []

  path.chars.each do |char|
    hex_code = char.ord.to_s(16)

    #arr << "char: '#{char}'; hex_code: #{hex_code}"
    #arr << "char: null; hex_code: 00"

    #arr << "'#{char}' : #{hex_code}"
    #arr << " : 00"

    arr << [ char, hex_code ]
    arr << [ nil, '00' ]
  end

  puts "result:"

  arr_1 = arr.map { |nested_arr| "'#{nested_arr[0]}':#{nested_arr[1]}" }
  puts arr_1.join(', ')
  puts

  arr_2 = arr.map { |nested_arr| nested_arr[1] }
  puts arr_2.join(' ')

  puts
  puts "occupied bytes count: #{arr.size}"

  arr
end
