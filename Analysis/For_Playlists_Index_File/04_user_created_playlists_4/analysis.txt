Now let's print the hex code of each character in the path. We will use the string including dots wherein for dot the hex code should be printed as 00.

```
def convert_given_path_characters_to_hex_code_separated_by_hexcode_00(path:)
  arr = []

  path.chars.each do |char|
    hex_code = char.ord.to_s(16)

    #arr << "char: '#{char}'; hex_code: #{hex_code}"
    #arr << "char: null; hex_code: 00"

    #arr << "'#{char}' : #{hex_code}"
    #arr << " : 00"

    arr << [ char, hex_code ]
    arr << [ nil, '00' ]
  end

  puts "result:"

  arr_1 = arr.map { |nested_arr| "'#{nested_arr[0]}':#{nested_arr[1]}" }
  puts arr_1.join(', ')
  puts

  arr_2 = arr.map { |nested_arr| nested_arr[1] }
  puts arr_2.join(' ')

  puts
  puts "occupied bytes count: #{arr.size}"

  arr
end
```
